Bash(git push * +*)
Bash(git push * --delete*)
Bash(git push * -d *)

# origin/HEAD; with no value the runner refuses to start). {merge_basename} is

Bash(git push * *:{default_branch})
Bash(git push * *:refs/heads/{default_branch})
